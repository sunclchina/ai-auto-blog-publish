# ============================================================================
# A-Blog 伴生服务 · Crontab 模板 (Linux)
# 归档位置: backend/crontab.txt (仓库内模板, 勿直接编辑部署机的副本)
# 部署方式: crontab -e 后粘贴本文件内容; 或 root 下复制为 /etc/cron.d/ablog
#
# ── 时区: Asia/Shanghai ──────────────────────────────────────────────────
# cron 按系统时区调度, 部署机务必先执行:
#     sudo timedatectl set-timezone Asia/Shanghai
# Debian/Ubuntu 的 Vixie-cron 还支持文件内 CRON_TZ 指令(本文件已启用),
# 即使系统时区是 UTC 也能按北京时间调度; 其他发行版以系统时区为准。
#
# ── 目录约定 (与 deploy/install.sh 一致, 默认安装到 /opt/ablog) ──────────
#   ABLOG_HOME = /opt/ablog                     # 安装根目录(装到别处就改下面变量)
#   Python     = ${ABLOG_HOME}/backend/venv/bin/python
#   日志       = ${ABLOG_HOME}/data/logs/
#
# ── 安全约定 ─────────────────────────────────────────────────────────────
#   - 命令行中绝不出现任何密钥; 密钥一律走 backend/.env (chmod 600,
#     systemd EnvironmentFile 注入), 见 deploy/security.md;
#   - 所有任务输出重定向到 data/logs/ 独立日志文件, 便于排查;
#   - 任务以非零码退出时 cron 会尝试投递 MAILTO, 建议填真实邮箱收告警。
#
# ── 调度入口契约 ─────────────────────────────────────────────────────────
# 以下调用 scheduler/daily_queue.py 的 CLI (后端实现, 参数含义见架构 4 节):
#   --generate  生成当日任务队列(栏目轮换/配额/时段随机分钟)
#   --topics    预选题: 对 queued 任务执行 Step1, 生成备选列表并选定默认选题
#               (任务保持 queued, 等待人工干预窗口或到时自动继续)
#   --run       执行 queued 任务(按 sort_order 顺序, Step2-7 生成正文)
#               → 发布到点任务 (REST 主通道 / XML-RPC 兜底)
#   --column X  只处理某栏目 (stock|tech|reading|book)
# ============================================================================

# ---- cron 环境 ----
MAILTO=""                                   # 建议填你的邮箱收失败告警; 空 = 不发送
CRON_TZ=Asia/Shanghai                       # 显式调度时区 (Vixie-cron)
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

ABLOG_HOME=/opt/ablog                       # ← 安装目录, 与 install.sh 保持一致
PY=${ABLOG_HOME}/backend/venv/bin/python
LOG=${ABLOG_HOME}/data/logs

# ---------------------------------------------------------------------------
# 1) 每日 20:00 触发 A股复盘任务 (仅交易日)
#    - crontab 只负责"按时触发"; 是否交易日由 Python 侧二次校验:
#      scheduler/calendar.py 内置 A股交易日历(周末 + 春节/国庆等法定节假日跳过),
#      非交易日直接退出, 不消耗任何 Token;
#    - 当日数据源失败(通达信/新浪/东财全挂)则当日跳过, 不补发、不重试;
#    - --column stock --run: 生成复盘任务并立即执行(受总开关/额度/熔断约束)。
# ---------------------------------------------------------------------------
0 20 * * *  cd ${ABLOG_HOME}/backend && ${PY} -m scheduler.daily_queue --column stock --run >> ${LOG}/cron-stock.log 2>&1

# ---------------------------------------------------------------------------
# 2) 每日 08:00 两阶段调度：先建队列 + 预选题 (Step1 备选列表)
#    - --generate  按轮换比例(复盘40/技术30/国学+书评30)生成当日任务队列,
#                  校验每日发文上限(1-10)与 Token 额度(200k), 在 09:00-21:00
#                  时段内为每篇分配随机分钟作为模拟人工发布时间;
#    - --topics    对 queued 任务执行 Step1 选题智能体: 采集素材 → 生成 3-5 条
#                  备选列表并选定默认选题, 任务保持 queued (不生成正文);
#    - 人工干预窗口 08:00-20:00 站长可在 WP 后台「智能选题中心」查看候选、
#                  指定/修改选题、删除任务、调整发布顺序 (↑↓);
#    - 无干预自动   20:00/20:30 的 --run 阶段自动继续: 已定选题直接采用
#                  (人工指定优先), 未干预任务自动用候选第 1 条, 无人值守。
# ---------------------------------------------------------------------------
0 8 * * *  cd ${ABLOG_HOME}/backend && ${PY} -m scheduler.daily_queue --generate --topics >> ${LOG}/cron-daily.log 2>&1

# ---------------------------------------------------------------------------
# 3) 每日 20:30 起执行队列 (生成正文 + 发布到点任务), 拆成三轮 (翁老: 批量拆小)
#    - batch.run_per_round 默认 1 篇/轮: 每轮最多生成 1 篇, 剩余任务下一轮继续,
#      避免集中生成打爆 WP/PHP 内存 (对应 2026-08-05 WP 内存耗尽事故);
#    - 复盘任务 20:00 已单独触发 (--column stock --run);
#    - 此处处理其余栏目: queued 任务按 sort_order 顺序生成正文 (Step2-7),
#      publish_date 到点即发布, 未到点自动落 WP future 定时发布;
#    - 幂等: 已 ready/published 任务自动跳过。三轮跑不完(如配置 run_per_round>1
#      或当日任务数更多)可自行追加 22:00 轮次。
# ---------------------------------------------------------------------------
30 20 * * *  cd ${ABLOG_HOME}/backend && ${PY} -m scheduler.daily_queue --run >> ${LOG}/cron-run.log 2>&1
0 21 * * *  cd ${ABLOG_HOME}/backend && ${PY} -m scheduler.daily_queue --run >> ${LOG}/cron-run.log 2>&1
30 21 * * *  cd ${ABLOG_HOME}/backend && ${PY} -m scheduler.daily_queue --run >> ${LOG}/cron-run.log 2>&1

# ---------------------------------------------------------------------------
# 4) 每 5 分钟健康检查 (可选)
#    - 探测本机 FastAPI /healthz, 失败仅写日志告警, 不负责拉起进程
#      (进程守护由 systemd ablog.service 的 Restart=on-failure 负责);
#    - 默认注释掉, 需要时删除行首 # 启用。
# ---------------------------------------------------------------------------
# */5 * * * *  curl -fsS http://127.0.0.1:8080/healthz >/dev/null 2>&1 || echo "[$(date '+\%F \%T')] /healthz FAIL" >> ${LOG}/cron-health.log 2>&1

# ---------------------------------------------------------------------------
# 5) 每日 03:00 备份 data/ablog.db (建议启用, 详见 deploy/security.md 第 5 节)
#    - 保留最近 30 天备份, 自动清理旧文件; 备份目录 ${ABLOG_HOME}/backup (chmod 700)。
# ---------------------------------------------------------------------------
# 0 3 * * *  cp ${ABLOG_HOME}/data/ablog.db ${ABLOG_HOME}/backup/ablog-$(date +\%F).db && find ${ABLOG_HOME}/backup -name 'ablog-*.db' -mtime +30 -delete >> ${LOG}/cron-backup.log 2>&1

# ============================================================================
# 维护速查:
#   手动补跑当日队列:  cd /opt/ablog/backend && venv/bin/python -m scheduler.daily_queue --generate --topics --run
#   确认 cron 已触发:   grep CRON /var/log/syslog | tail -20
#   服务健康:           curl http://127.0.0.1:8080/healthz
#   改用 /etc/cron.d/ablog 时: 每条命令前须加用户字段(如 ablog),
#   且该文件属主必须是 root, 否则 cron 静默忽略。
